cmake_minimum_required(VERSION 3.20)
project(AttentionPPOParity LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Torch REQUIRED)
add_executable(attention_ppo_parity PPOParity.cpp)
separate_arguments(TORCH_BUILD_OPTIONS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(attention_ppo_parity PRIVATE ${TORCH_BUILD_OPTIONS})
target_link_libraries(attention_ppo_parity PRIVATE "${TORCH_LIBRARIES}")
if(MSVC)
 target_compile_options(attention_ppo_parity PRIVATE /bigobj)
endif()
