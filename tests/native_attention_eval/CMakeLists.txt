cmake_minimum_required(VERSION 3.20)
project(NativeAttentionEvalTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Torch REQUIRED)
if(NOT GIGALEARN_SOURCE)
  message(FATAL_ERROR "Set GIGALEARN_SOURCE")
endif()
add_subdirectory("${GIGALEARN_SOURCE}/GigaLearnCPP/RLGymCPP" rlgym)
add_executable(attention_eval_smoke Smoke.cpp)
target_link_libraries(attention_eval_smoke PRIVATE RLGymCPP "${TORCH_LIBRARIES}")
target_include_directories(attention_eval_smoke PRIVATE "${GIGALEARN_SOURCE}/GigaLearnCPP/libsrc/json" ${TORCH_INCLUDE_DIRS})
separate_arguments(TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(attention_eval_smoke PRIVATE ${TORCH_FLAGS})
target_compile_definitions(attention_eval_smoke PRIVATE ATTENTION_SOURCE_SHA="native-eval-test-contract")
if(MSVC)
 target_compile_options(attention_eval_smoke PRIVATE /bigobj)
endif()
