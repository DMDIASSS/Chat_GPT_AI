cmake_minimum_required(VERSION 3.24)
project(GigaForge LANGUAGES CXX)
if(WIN32)
 add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Torch REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
set(GGL_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/vendor" CACHE PATH "Pinned GigaLearn source")
set(LIBRARY_OUTPUT_PATH "${CMAKE_BINARY_DIR}")
set(EXECUTABLE_OUTPUT_PATH "${CMAKE_BINARY_DIR}")
add_subdirectory("${GGL_VENDOR}/GigaLearnCPP" giga)
add_executable(GigaForge src/main.cpp)
target_include_directories(GigaForge PRIVATE "${GGL_VENDOR}/GigaLearnCPP/src/private" "${GGL_VENDOR}/GigaLearnCPP/RLGymCPP/RocketSim/src")
target_link_libraries(GigaForge PRIVATE GigaLearnCPP "${TORCH_LIBRARIES}")
if(MSVC)
 target_compile_options(GigaForge PRIVATE /bigobj /utf-8)
endif()
