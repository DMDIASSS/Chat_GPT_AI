cmake_minimum_required(VERSION 3.20)
project(AttentionForge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python 3.11 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
set(ROCKETSIM_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/RocketSim-c2baacb8f4b441dd8505e63c2aeb5a1679b60b02" CACHE PATH "Pinned RocketSim checkout")
add_subdirectory("${ROCKETSIM_SOURCE}" rocketsim)
pybind11_add_module(_attention_native native/core.cpp)
target_include_directories(_attention_native PRIVATE "${ROCKETSIM_SOURCE}/src")
target_link_libraries(_attention_native PRIVATE RocketSim)
if(MSVC)
 target_compile_options(_attention_native PRIVATE /O2 /EHsc)
 target_compile_options(RocketSim PRIVATE /O2 /EHsc)
else()
 target_compile_options(_attention_native PRIVATE -O3)
 target_compile_options(RocketSim PRIVATE -O3)
endif()
install(TARGETS _attention_native LIBRARY DESTINATION attentionforge RUNTIME DESTINATION attentionforge)
