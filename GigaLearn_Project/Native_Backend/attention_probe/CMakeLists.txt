cmake_minimum_required(VERSION 3.20)
project(AttentionNativeProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Torch REQUIRED)
add_executable(attention_native_probe AttentionProbe.cpp)
separate_arguments(TORCH_BUILD_OPTIONS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(attention_native_probe PRIVATE ${TORCH_BUILD_OPTIONS})
target_link_libraries(attention_native_probe PRIVATE "${TORCH_LIBRARIES}")

file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/AttentionProbe.cpp" ATTENTION_PROBE_SHA)
target_compile_definitions(attention_native_probe PRIVATE ATTENTION_PROBE_SHA="${ATTENTION_PROBE_SHA}")
if(MSVC)
 target_compile_options(attention_native_probe PRIVATE /bigobj)
endif()
