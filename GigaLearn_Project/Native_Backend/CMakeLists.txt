cmake_minimum_required(VERSION 3.20)
project(SlaterGigaNative LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 20)
if(NOT GIGALEARN_SOURCE OR NOT EXISTS "${GIGALEARN_SOURCE}/GigaLearnCPP/CMakeLists.txt")
 message(FATAL_ERROR "Set GIGALEARN_SOURCE to the isolated pinned checkout")
endif()
add_subdirectory("${GIGALEARN_SOURCE}/GigaLearnCPP" giga)
find_package(Torch REQUIRED)
add_executable(SlaterGigaNative NativeMain.cpp)
# ZIP extraction can preserve older timestamps than an existing object file.
# A content-derived compile flag makes Ninja rebuild changed backend code anyway.
file(GLOB SLATER_BACKEND_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
list(SORT SLATER_BACKEND_FILES)
set(SLATER_BACKEND_DIGESTS "")
foreach(SLATER_BACKEND_FILE IN LISTS SLATER_BACKEND_FILES)
 file(SHA256 "${SLATER_BACKEND_FILE}" SLATER_BACKEND_FILE_SHA)
 string(APPEND SLATER_BACKEND_DIGESTS "${SLATER_BACKEND_FILE_SHA}")
endforeach()
string(SHA256 SLATER_BACKEND_SHA "${SLATER_BACKEND_DIGESTS}")
target_compile_definitions(SlaterGigaNative PRIVATE SLATER_BACKEND_SOURCE_SHA="${SLATER_BACKEND_SHA}")
separate_arguments(TORCH_BUILD_OPTIONS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(SlaterGigaNative PRIVATE ${TORCH_BUILD_OPTIONS})
target_link_libraries(SlaterGigaNative PRIVATE GigaLearnCPP "${TORCH_LIBRARIES}")
target_include_directories(SlaterGigaNative PRIVATE ${TORCH_INCLUDE_DIRS} "${GIGALEARN_SOURCE}/GigaLearnCPP/src/private")
if(MSVC)
 target_compile_options(SlaterGigaNative PRIVATE /bigobj)
 file(GLOB TORCH_DLLS "${TORCH_INSTALL_PREFIX}/lib/*.dll")
 add_custom_command(TARGET SlaterGigaNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TORCH_DLLS} $<TARGET_FILE_DIR:SlaterGigaNative>)
 add_custom_command(TARGET SlaterGigaNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:GigaLearnCPP> $<TARGET_FILE_DIR:SlaterGigaNative>)
endif()
