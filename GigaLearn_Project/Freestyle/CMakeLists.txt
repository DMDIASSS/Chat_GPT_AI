cmake_minimum_required(VERSION 3.20)
project(SlaterFreestyleNative LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 20)
if(NOT GIGALEARN_SOURCE OR NOT EXISTS "${GIGALEARN_SOURCE}/GigaLearnCPP/CMakeLists.txt")
 message(FATAL_ERROR "Set GIGALEARN_SOURCE to the isolated verified GigaLearn checkout")
endif()
add_subdirectory("${GIGALEARN_SOURCE}/GigaLearnCPP" giga)
find_package(Torch REQUIRED)
# GigaLearn has a per-step callback but no callback after PPO finishes. Generate
# a Freestyle-only translation unit with a complete Report hook, without editing
# the pinned shared upstream checkout or changing the existing Slater trainers.
set(ORIGINAL_LEARNER "${GIGALEARN_SOURCE}/GigaLearnCPP/src/public/GigaLearnCPP/Learner.cpp")
file(READ "${ORIGINAL_LEARNER}" FREESTYLE_LEARNER)
set(REPORT_ANCHOR "report.Finish();")
string(REGEX MATCHALL "report\\.Finish\\(\\)" REPORT_ANCHORS "${FREESTYLE_LEARNER}")
list(LENGTH REPORT_ANCHORS REPORT_ANCHOR_COUNT)
if(NOT REPORT_ANCHOR_COUNT EQUAL 2)
 message(FATAL_ERROR "Unexpected pinned Learner.cpp Report hook count; refusing unverified source")
endif()
string(REPLACE "using namespace RLGC;" "using namespace RLGC;\nextern \"C\" void SlaterFreestyleReportHook(GGL::Learner*, const GGL::Report&);\nextern \"C\" void SlaterFreestyleCheckpointHook(GGL::Learner*);" FREESTYLE_LEARNER "${FREESTYLE_LEARNER}")
string(REPLACE "${REPORT_ANCHOR}" "${REPORT_ANCHOR}\n                SlaterFreestyleReportHook(this, report);" FREESTYLE_LEARNER "${FREESTYLE_LEARNER}")
set(STOP_ANCHOR "std::cout << \"SLATER_GIGA_COMPLETE iterations=\" << totalIterations << \" timesteps=\" << totalTimesteps << std::endl;\n                    exit(0);")
string(FIND "${FREESTYLE_LEARNER}" "${STOP_ANCHOR}" STOP_INDEX)
if(STOP_INDEX EQUAL -1)
 message(FATAL_ERROR "Pinned source must have the verified bounded-training compatibility patch")
endif()
string(REPLACE "${STOP_ANCHOR}" "std::cout << \"SLATER_GIGA_COMPLETE iterations=\" << totalIterations << \" timesteps=\" << totalTimesteps << std::endl;\n                    return;" FREESTYLE_LEARNER "${FREESTYLE_LEARNER}")
set(SAVE_ANCHOR "if (versionMgr)\n\t\tversionMgr->SaveVersions();")
string(FIND "${FREESTYLE_LEARNER}" "${SAVE_ANCHOR}" SAVE_INDEX)
if(SAVE_INDEX EQUAL -1)
 message(FATAL_ERROR "Unexpected pinned save implementation")
endif()
string(REPLACE "${SAVE_ANCHOR}" "${SAVE_ANCHOR}\n    SlaterFreestyleCheckpointHook(this);" FREESTYLE_LEARNER "${FREESTYLE_LEARNER}")
# Incomplete writes are preserved on disk but can never become the resume source.
set(LOAD_ANCHOR "for (int64_t timesteps : allSavedTimesteps)\n\t\thighest = RS_MAX(timesteps, highest);")
string(FIND "${FREESTYLE_LEARNER}" "${LOAD_ANCHOR}" LOAD_INDEX)
if(LOAD_INDEX EQUAL -1)
 message(FATAL_ERROR "Unexpected pinned checkpoint selection implementation")
endif()
string(REPLACE "${LOAD_ANCHOR}" "for (int64_t timesteps : allSavedTimesteps) {\n        if (!std::filesystem::exists(config.checkpointFolder / std::to_string(timesteps) / \"FREESTYLE_COMPLETE.json\")) continue;\n        highest = RS_MAX(timesteps, highest);\n    }" FREESTYLE_LEARNER "${FREESTYLE_LEARNER}")
string(REPLACE "std::filesystem::create_directories(saveFolder);" "std::filesystem::create_directories(saveFolder);\n    std::filesystem::remove(saveFolder / \"FREESTYLE_COMPLETE.json\");" FREESTYLE_LEARNER "${FREESTYLE_LEARNER}")
# Freestyle trains only the blue learner. Orange rows are controlled by the
# optional deterministic goalie and must never enter PPO/log-prob trajectories.
set(PLAYER_ANCHOR "for (int i = 0; i < numPlayers; i++)\n\t\t\t\tnewPlayerIndices.push_back(i);")
string(FIND "${FREESTYLE_LEARNER}" "${PLAYER_ANCHOR}" PLAYER_INDEX)
if(PLAYER_INDEX EQUAL -1)
 message(FATAL_ERROR "Unexpected controlled-player collection source")
endif()
string(REPLACE "${PLAYER_ANCHOR}" "int playerIndex=0;\n            for(const auto& state: envSet->state.gameStates) {\n                for(const auto& player: state.players) {\n                    if(player.team == Team::BLUE) newPlayerIndices.push_back(playerIndex);\n                    ++playerIndex;\n                }\n            }\n            tNewPlayerIndices=torch::tensor(newPlayerIndices);" FREESTYLE_LEARNER "${FREESTYLE_LEARNER}")
string(REPLACE "int numRealPlayers = oldVersion ? newPlayerIndices.size() : envSet->state.numPlayers;" "int numRealPlayers = newPlayerIndices.size();" FREESTYLE_LEARNER "${FREESTYLE_LEARNER}")
set(INFER_ANCHOR "torch::Tensor tdStates = tStates.to(ppo->device, true);\n\t\t\t\t\t\t\ttorch::Tensor tdActionMasks = tActionMasks.to(ppo->device, true);\n\t\t\t\t\t\t\tppo->InferActions(tdStates, tdActionMasks, &tActions, &tLogProbs);\n\t\t\t\t\t\t\ttActions = tActions.cpu();")
string(FIND "${FREESTYLE_LEARNER}" "${INFER_ANCHOR}" INFER_INDEX)
if(INFER_INDEX EQUAL -1)
 message(FATAL_ERROR "Unexpected learner inference source; cannot guarantee defender exclusion")
endif()
string(REPLACE "${INFER_ANCHOR}" "torch::Tensor tdStates = tStates.index_select(0,tNewPlayerIndices).to(ppo->device, true);\n                            torch::Tensor tdActionMasks = tActionMasks.index_select(0,tNewPlayerIndices).to(ppo->device,true);\n                            torch::Tensor tLearnerActions;\n                            ppo->InferActions(tdStates,tdActionMasks,&tLearnerActions,&tLogProbs);\n                            tActions=torch::zeros({numPlayers},tLearnerActions.options().device(torch::kCPU));\n                            tActions.index_copy_(0,tNewPlayerIndices,tLearnerActions.cpu());" FREESTYLE_LEARNER "${FREESTYLE_LEARNER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/FreestyleLearner.cpp" "${FREESTYLE_LEARNER}")
get_target_property(GIGA_SOURCES GigaLearnCPP SOURCES)
list(REMOVE_ITEM GIGA_SOURCES "${ORIGINAL_LEARNER}")
set_property(TARGET GigaLearnCPP PROPERTY SOURCES ${GIGA_SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/FreestyleLearner.cpp")
target_include_directories(GigaLearnCPP PRIVATE "${GIGALEARN_SOURCE}/GigaLearnCPP/src/public/GigaLearnCPP")
# Link the full-report hook inside the shared library itself, avoiding Windows
# executable-symbol resolution. It calls back through a registered function.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/FreestyleHook.cpp" "#include <GigaLearnCPP/Learner.h>\nnamespace { void (*callback)(GGL::Learner*, const GGL::Report&) = nullptr; void (*saveCallback)(GGL::Learner*)=nullptr; }\nextern \"C\" RG_IMEXPORT void SlaterFreestyleSetReportHook(void (*fn)(GGL::Learner*, const GGL::Report&)) { callback=fn; }\nextern \"C\" RG_IMEXPORT void SlaterFreestyleSetCheckpointHook(void (*fn)(GGL::Learner*)) { saveCallback=fn; }\nextern \"C\" void SlaterFreestyleReportHook(GGL::Learner* learner, const GGL::Report& report) { if(callback) callback(learner, report); }\nextern \"C\" void SlaterFreestyleCheckpointHook(GGL::Learner* learner) { if(saveCallback) saveCallback(learner); }\n")
target_sources(GigaLearnCPP PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/FreestyleHook.cpp")
add_executable(SlaterFreestyleNative FreestyleMain.cpp)
file(GLOB FREESTYLE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT FREESTYLE_FILES)
set(FREESTYLE_DIGESTS "")
foreach(FREESTYLE_FILE IN LISTS FREESTYLE_FILES)
 file(SHA256 "${FREESTYLE_FILE}" FREESTYLE_SHA)
 string(APPEND FREESTYLE_DIGESTS "${FREESTYLE_SHA}")
endforeach()
string(SHA256 FREESTYLE_SOURCE_SHA "${FREESTYLE_DIGESTS}")
target_compile_definitions(SlaterFreestyleNative PRIVATE FREESTYLE_SOURCE_SHA="${FREESTYLE_SOURCE_SHA}")
separate_arguments(TORCH_OPTIONS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(SlaterFreestyleNative PRIVATE ${TORCH_OPTIONS})
target_link_libraries(SlaterFreestyleNative PRIVATE GigaLearnCPP "${TORCH_LIBRARIES}")
target_include_directories(SlaterFreestyleNative PRIVATE ${TORCH_INCLUDE_DIRS} "${GIGALEARN_SOURCE}/GigaLearnCPP/src/private")
if(TORCH_CUDA_LIBRARIES)
 target_compile_definitions(SlaterFreestyleNative PRIVATE FREESTYLE_HAS_CUDA)
endif()
if(MSVC)
 target_compile_options(SlaterFreestyleNative PRIVATE /bigobj)
 file(GLOB TORCH_DLLS "${TORCH_INSTALL_PREFIX}/lib/*.dll")
 add_custom_command(TARGET SlaterFreestyleNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TORCH_DLLS} $<TARGET_FILE_DIR:SlaterFreestyleNative>)
 add_custom_command(TARGET SlaterFreestyleNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:GigaLearnCPP> $<TARGET_FILE_DIR:SlaterFreestyleNative>)
endif()
