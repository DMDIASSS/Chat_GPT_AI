cmake_minimum_required(VERSION 3.20)
project(SlaterNativeAttention LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT GIGALEARN_SOURCE OR NOT EXISTS "${GIGALEARN_SOURCE}/GigaLearnCPP/RLGymCPP/CMakeLists.txt")
 message(FATAL_ERROR "Provide the verified pinned GigaLearn source checkout")
endif()
# Reuse the mature simulation environment. The flat GigaLearn policy cannot
# directly own this TorchScript attention actor; its PPO adapter is separate.
add_subdirectory("${GIGALEARN_SOURCE}/GigaLearnCPP/RLGymCPP" rlgym)
find_package(Torch REQUIRED)
set(ATTENTION_JSON "${GIGALEARN_SOURCE}/GigaLearnCPP/libsrc/json")
file(GLOB ATTENTION_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
list(SORT ATTENTION_SOURCES)
set(ATTENTION_DIGESTS "")
foreach(SOURCE IN LISTS ATTENTION_SOURCES)
 file(SHA256 "${SOURCE}" DIGEST)
 string(APPEND ATTENTION_DIGESTS "${DIGEST}")
endforeach()
string(SHA256 ATTENTION_SOURCE_SHA "${ATTENTION_DIGESTS}")
separate_arguments(ATTENTION_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
add_executable(SlaterAttentionNative AttentionMain.cpp)
target_link_libraries(SlaterAttentionNative PRIVATE RLGymCPP "${TORCH_LIBRARIES}")
target_include_directories(SlaterAttentionNative PRIVATE "${ATTENTION_JSON}" ${TORCH_INCLUDE_DIRS})
target_compile_options(SlaterAttentionNative PRIVATE ${ATTENTION_TORCH_FLAGS})
target_compile_definitions(SlaterAttentionNative PRIVATE ATTENTION_SOURCE_SHA="${ATTENTION_SOURCE_SHA}")
add_executable(SlaterAttentionObsParity ObsParity.cpp)
target_link_libraries(SlaterAttentionObsParity PRIVATE RLGymCPP)
target_include_directories(SlaterAttentionObsParity PRIVATE "${ATTENTION_JSON}")
target_compile_definitions(SlaterAttentionObsParity PRIVATE ATTENTION_SOURCE_SHA="${ATTENTION_SOURCE_SHA}")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/native_attention_ppo/PPOParity.cpp")
 add_executable(SlaterAttentionPPOParity "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/native_attention_ppo/PPOParity.cpp")
 target_link_libraries(SlaterAttentionPPOParity PRIVATE "${TORCH_LIBRARIES}")
 target_include_directories(SlaterAttentionPPOParity PRIVATE "${ATTENTION_JSON}" ${TORCH_INCLUDE_DIRS})
 target_compile_options(SlaterAttentionPPOParity PRIVATE ${ATTENTION_TORCH_FLAGS})
 target_compile_definitions(SlaterAttentionPPOParity PRIVATE ATTENTION_SOURCE_SHA="${ATTENTION_SOURCE_SHA}")
endif()
if(MSVC)
 target_compile_options(SlaterAttentionNative PRIVATE /bigobj)
 if(TARGET SlaterAttentionPPOParity)
  target_compile_options(SlaterAttentionPPOParity PRIVATE /bigobj)
 endif()
endif()
